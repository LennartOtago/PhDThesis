\section{Correlation between Latent Field and Hyper-Parameter}
\label{ap:Correlation}
In the book Gaussian Markov Random Fields~\cite{rue2005gaussian}, Rue and Held demonstrate that a strong correlation between the hyper-parameter $\mu$ and the latent field $\bm{x}$ can significantly slow down convergence particularly when using Gibbs samplers.  
They consider the hierarchical model
\begin{subequations}
	\label{eq:rueMod}
	\begin{align}
		\mu &\sim \mathcal{N}(0, 1) \\
		\bm{x} | \mu &\sim \mathcal{N}(\mu \bm{1}, \bm{Q}^{-1}),
	\end{align}
\end{subequations}
and apply a Gibbs sampler based on the full conditional distributions
\begin{align}
	\mu^{(k)} | \bm{x}^{(k)} &\sim \mathcal{N} \left( \frac{\bm{1}^T \bm{Q} \bm{x}^{(k-1)}}{1 + \bm{1}^T \bm{Q} \bm{1}}, \, \left(1 + \bm{1}^T \bm{Q} \bm{1} \right)^{-1} \right) \label{eq:gibbsMu} \\
	\bm{x}^{(k)} | \mu^{(k)} &\sim \mathcal{N}(\mu^{(k)} \bm{1}, \bm{Q}^{-1})\label{eq:gibbsx}.
\end{align}

As illustrated in Figure~\ref{fig:RueHeld}, when the sampler is restricted to steps only in the $\mu$-direction (horizontal axis) or the $\bm{x}$-direction (vertical axis), it requires many iterations to adequately explore the parameter space. 
This inefficiency arises from the high correlation between $\mu$ and $\bm{x}$, visible in Figure~\ref{fig:RueHeld} as a ``squeeze'' of the distribution.
\begin{figure}[ht!]
	\centering
	\includegraphics[width = \textwidth]{Figures/RueHeldBookFig.png}
	\caption[Correlation structure in between parameters and hyper-parameters]{The figure taken from~\cite[Figure 4.1 (b)]{rue2005gaussian} displays samples $\mu^{(k)}$ on the horizontal axis (x-axis) and $\bm{1}^T \bm{Q} \bm{x}^{(k)}$ on the vertical axis (y-axis) from the hierarchical model in Eq.~\ref{eq:rueMod} for over 1000 iterations.
		The algorithm updates $\mu$ and $\bm{x}$ successively from their full conditional distributions (see Eq.~\ref{eq:gibbsMu} and Eq.~\ref{eq:gibbsx}). The slow mixing is due to the strong correlation of $\mu$ and $\bm{x}$ (see full conditionals).
		The sampler permits only axis-aligned (horizontal and vertical) moves and does not allow diagonal moves, as illustrated by the arrows.}
	\label{fig:RueHeld}
\end{figure}

A solution to the slow mixing problem is to update $(\mu, \bm{x})$ jointly.
Since here $\mu$ is one-dimensional, effectively only the marginal density of $\mu$ is needed.
\begin{align}
	\mu^{\star}  &\sim q (\mu^{\star}|	\mu^{(k-1)} ) \\
	\bm{x}^{(k)} | \mu^{\star} &\sim \mathcal{N} (	\mu^{\star}\bm{1}, \bm{Q}^{-1}) 
\end{align}
With a simple MCMC algorithm targeting $ \mu$ one can explore the sample space (of $ \mu$) efficiently and only draw a sample for $\bm{x}$ from its full conditional once the proposal $\mu^{\star} =\mu^{(k)} $ has been accepted.

\section{Monte-Carlo Error and Integrated Autocorrelation Time}
\label{ap:IATC}
To assess the error $(\sigma^{(i)})^2$ of a samples-based estimate 
\begin{align}
	\bar{h}_N^{(i)} \coloneqq	\text{E}_{\bm{x}|\bm{y}} [h(\bm{x})] = \frac{1}{N} \sum_{k=1}^{N} h(\bm{x}^{(k)}),
\end{align} 
from the chain $\mathcal{M}^{(i)} = \{\bm{x}^{(1)}, \dots,\bm{x}^{(k)},\dots, \bm{x}^{(s)},\dots, \bm{x}^{(N)}\}$, where $\mathcal{M}^{(i)} \sim \pi(\bm{x}|\bm{y})$, we ignore systematic error due to initialisation bias (burn-in period), but we have to take into account that samples produced by any system or algorithm are correlated.
To derive the IACT, we follow Ulli Wolff's lecture notes \cite{wolff2002LecNot} (or alternatively \cite{wolff2004monte}).

In general, the error of a Monte-Carlo estimate is:
\begin{align}
	(\sigma^{(i)})^2 = \text{Var}(\bar{h}_N^{(i)}) =  \text{Var}(\text{E}_{\bm{x}|\bm{y}} [h(\bm{x})]) = \Bigg( \frac{1}{N} \sum_{k=1}^{N} h(\bm{x}^{(k)}) - \bar{h}_N^{(i)} \Bigg)^2 \, .
\end{align}
Expanding this summation, the error becomes
\begin{align}
	(\sigma^{(i)})^2 = \frac{1}{N^2} \sum_{k,s=1}^{N} \Gamma(k-s)
\end{align}
with the autocorrelation coefficient $\Gamma(k-s) =  \big( h(\bm{x}^{(k)}) -\bar{h}_N^{(i)} \big) \big(h(\bm{x}^{(s)}) - \bar{h}_N^{(i)}\big)$.
Next we rewrite
\begin{align}
	\sum_{k,s=1}^{N} \Gamma(k-s) = \text{Var}(h(\bm{x}))  \sum_{k,s=1}^{N} \frac{\Gamma(k-s)}{\Gamma(0)} =  \text{Var}(h(\bm{x})) \sum_{k,s=1}^{N}\rho(k-s)\, ,
\end{align}
with the normalised autocorrelation coefficient $\rho(k-s) =  \Gamma(k-s)/ \Gamma(0)$ at lag $k-s$ and $\Gamma(0) = \text{Var}(h(\bm{x}) )$ for $k=s$.
Typically $\Gamma(t)$ decays exponentially so that, for $N\gg \tau$, $\Gamma(t) \overset{t \rightarrow \infty }{ \propto} \exp\{ - |t| / \tau \}  $ and one can approximate
\begin{align}
	\sum_{k,s=1}^{N}\rho(k-s)  = N \sum_{t = -(N-1) }^{N-1} \Bigg(1- \frac{t}{N} \Bigg) \rho(t)  \approx N  \sum_{t = - \infty }^{\infty} \rho(t) \, ,
\end{align}
see \cite[p. 137]{Sokal1997}.
If $\tau \gg 1$
\begin{align}
	\sum_{t = - \infty }^{\infty} \rho(t) =  1 + 2 \sum_{t = 1}^{\infty} \big(e^{-1/ \tau}\big)^t =  1 + 2 \frac{e^{-1/ \tau} }{1 - e^{-1/ \tau}} \approx  1 + 2 \frac{1 -1/ \tau }{1/ \tau} =  2 \tau -1 \, .%\approx 2 \tau_{int}\, ,
\end{align}
%so that $\tau_{\text{int}} \approx \tau$.
Here the geometric power series $\sum^{\infty}_{n=0} x^n= 1/ (1-x)$ and the Taylor series $ e^x \approx 1+x$ for small $x$ is used.
In practice, the estimate for the Monte-Carlo error is:
\begin{align}
	(\sigma^{(i)})^2   \approx \frac{\text{Var}(h(\bm{x}) )}{N} \sum_{t = - \infty }^{\infty} \rho(t)
	\approx \frac{\text{Var}(h(\bm{x}) )}{N} \Bigg( \underbrace{  1 + 2 \sum_{t = 1}^{W} \rho(t)  }_{ \coloneqq 	\tau_{\text{int}} }\Bigg) = \text{Var}(h(\bm{x})) \frac{ \tau_{\text{int}} }{N}\, ,
\end{align}
where $W$ is the summation window and we define the IACT as twice the value as in \cite[pp. 103-105]{wolff2002LecNot}.
The IACT provides a good estimate of how many steps the sampling algorithm needs to take to produce one independent sample.
More specifically, the effective sample size $\frac{N}{ \tau_{\text{int}} }$ gives an estimate of how efficient a sampler is.

\begin{mccorrection}
	At the time this code was developed, I was not aware of an existing Python implementation of the SIRT framework that was under active development~\cite{alexDeepPy}. As a result, the algorithms used in this thesis were implemented independently and tailored specifically to the inverse problems considered here.
	An alternative MATLAB implementation is available through the TT-Toolbox~\cite{ttToolbox}.
	
	The code focuses primarily on efficient posterior approximation and sampling from TT representations. In particular, the implementation of the TT-cross approximation using the \texttt{ttpy} package~\cite{Oseledets2018ttpy} resulted in a substantial reduction in runtime compared to the PyTorch implementation of the SIRT framework by A. de Beer~\cite{alexDeepPy} for the test problems considered in this thesis.
	
	Furthermore, the implementation presented here can compute marginal distributions directly, a feature that is currently not available in \cite{alexDeepPy}. Inspecting the marginal distributions provides a useful diagnostic tool for assessing the quality of the TT approximation. If the marginal distributions show unexpected behaviour, this may indicate that the TT approximation is inaccurate or that the chosen computational domain is inadequate.
	In addition, the possibility of specifying individual TT ranks and initialising the TT approximation from a previous approximation was not available in earlier versions of~\cite{alexDeepPy} and, to the best of my knowledge, is currently under development. These features are important because they allow the approximation to exploit known problem structures, such as localised correlations, and can substantially reduce the number of function evaluations required to construct an accurate TT approximation.
	
	However, compared to~\cite{alexDeepPy}, this implementation currently lacks some more general-purpose features, such as changing the reference measure, using alternative interpolation bases, performing deep SIRT~\cite{cui2022deep}, or estimating normalising constants.
	
	To install the \texttt{ttpy} package, I found \newline
	\texttt{pip install git+https://github.com/oseledets/ttpy}
	to be the most reliable approach. Alternatively, the package can be run via the Docker-Image
	\texttt{daskol/ttpy}. 
	The package can then be imported using either
	\texttt{import tt} or \texttt{import ttpy}.
	To illustrate the workflow and potential reuse of the code, we provide an example demonstrating how samples and marginal distributions can be generated from a TT approximation.
\end{mccorrection}

def MargBack(TTCore, univarGrid):
	''' backward marginalisation (see "Prop.~\ref{prob:backMarg}") as in SIRT from Cui and Dolgov"~\cite{cui2022deep}" '''
	
	dim = len(univarGrid)
	B = dim * [None]  # coeffTensor
	B[-1] = TTCore[-1]
	R = [None] * dim
	C = [None] * dim
	
	# compute backward marginalisation coefficients sarting at k = dim - 1
	for k in range(dim - 1, 0, -1):
		r_kmin1, n, r_k = np.shape(TTCore[k])
		# "Eq.~\ref{eq:MassMat}, \cite[Eq.~22]{cui2022deep}" !! we set Lebesgue Measure to const = one
		M = np.identity(n) * (univarGrid[k][-1] - univarGrid[k][0])  # Mass matrix
		L = np.linalg.cholesky(M)
		
		# construct Tensor C  "Eq.~\ref{eq:constrCBack}, \cite[Eq.~27]{cui2022deep}"
		# C[k] = np.zeros((r_kmin1, n, r_k))
		# for alpha in range(0, r_kmin1):
		#     for l in range(0, r_k):
		#         C[k][alpha, :, l] = B[k][alpha, :, l] @ L[:, :]
		C[k] = np.einsum('jbi,kl->jbi', B[k], L)

		# unfold along first coordinate and compute thin QR decomposition of C^T "Eq.~\ref{eq:thinQRBack}, \cite[Eq.~28]{cui2022deep}"
		Q, R[k] = np.linalg.qr(C[k].reshape((r_kmin1, n * r_k), order='C').transpose(), mode='reduced')
		
		# compute next coefficient tensor "Eq.~\ref{eq:nextCoeffTBack}, \cite[Eq.~29]{cui2022deep}"
		# r_kmin2, n, r_kmin1 = np.shape(TTCore[k - 1])
		# B[k - 1] = np.zeros(np.shape(TTCore[k - 1]))
		# for alpha_2 in range(0, r_kmin2):
		#     for l_1 in range(0, r_kmin1):
		#         B[k - 1][alpha_2, :, l_1] = TTCore[k - 1][alpha_2, :, :] @ R[k][l_1, :]
		B[k - 1] = np.einsum('jbi,ki->jbk', TTCore[k - 1], R[k])

	''' forward marginalisation (see "Prop.~\ref{prob:ForMarg}") 
		similar to backward marginalsiation as in Cui and Dolgov "\cite{cui2022deep}" '''
	

	dim = len(univarGrid)
	Bpre = dim * [None]  # coeffTensor
	Rpre = [None] * dim
	Cpre = [None] * dim
	
	# initialise Bpre
	Bpre[0] = TTCore[0]
	
	# compute forward marginalisation coefficients sarting at k = 0
	for k in range(0, dim-1):
		r_kmin1, n, r_k = np.shape(TTCore[k])
		#  "Eq.~\ref{eq:MassMat}, \cite[Eq.~22]{cui2022deep}" !! we set Lebesgue Measure to const = one
		M = np.identity(n) * (univarGrid[k][-1] - univarGrid[k][0])  # Mass matrix
		L = np.linalg.cholesky(M)
		
		# construct Tensor C  "Eq.~\ref{eq:constrCForw}"
		# Cpre[k] = np.zeros((r_kmin1, n, r_k))
		# for alpha in range(0, r_kmin1):
		#     for l in range(0, r_k):
		#         Cpre[k][alpha, :, l] = Bpre[k][alpha, :, l] @ L[:, :]
		Cpre[k] = np.einsum( 'ail,it->atl', Bpre[k], L)
		
		# unfold along first coordinate and compute thin QR decomposition of C as in "Eq.~\ref{eq:thinQRForw}"
		Q, Rpre[k] = np.linalg.qr(Cpre[k].reshape((r_kmin1 * n, r_k), order='C'), mode='reduced')
		
		# compute next coefficient tensor  "Eq.~\ref{eq:nextCoeffTForw}"
		# r_k, n, r_kpls1 = np.shape(TTCore[k + 1])
		# Bpre[k + 1] = np.zeros(np.shape(TTCore[k + 1]))

		#         Bpre[k + 1][l_1, :, alpha_1] = Rpre[k][l_1,:] @ TTCore[k + 1][:, :, alpha_1]
		Bpre[k + 1] = np.einsum('lj,jbi->lbi', Rpre[k], TTCore[k + 1])
		
	return Rpre, Bpre

\begin{lstlisting}[language=Python,	 numbers=none, columns=flexible, caption = {Python code to draw samples via the SIRT scheme as in Alg.~Box~\ref{alg:SIRT}.}, escapeinside={"}{"}]
def SIRT(seeds, SQTT, univarGrid, B, absError):
	''' do squared inverse rosenblatt transform (SIRT) as in "Alg.~Box~\ref{alg:SIRT}" and "\cite{cui2022deep}" '''
	
	dim, numbSampl = seeds.shape
	sampls = np.zeros(seeds.shape) # samples from approximated PDF
	probVal = np.zeros(seeds.shape) # 'cond. marg.' PDF values, for MH-correction step
	Approx = np.zeros(seeds.shape[1]) # TT-Approx., to compare to true function
	
	# Lebesgue measure for quadrature "Eq.~\ref{eq:lebesgueWeight}"
	WholeLebLam = np.zeros(dim)
	for k in range(0, dim):
		WholeLebLam[k] = (univarGrid[k][-1] - univarGrid[k][0])
	lamX = np.ones(dim)
	for k in range(1, dim):
		lamX[k - 1] = np.prod(WholeLebLam[k:])
	
	gamError = absError / np.prod(WholeLebLam) # error as in "Eq.~\ref{eq:gamErr} \cite[Eq.~35]{cui2022deep}"
	
	# sample from first dimension "\cite[Eq.~30]{cui2022deep}"
	firstMarg = gamError * lamX[0] + np.sum(B[0][0, :, :] ** 2, 1)
	# cumulative distribution function, normalised numerically "Eq.~\ref{eq:CurrCDF}~\cite[Eq.~17]{cui2022deep}"
	firstCDF = scy.integrate.cumulative_trapezoid(firstMarg, univarGrid[0], initial=0)
	# draw samples as 'inverse transform'
	sampls[0] = np.interp(seeds[0], firstCDF / firstCDF[-1], univarGrid[0])
	probVal[0] = np.interp(sampls[0], univarGrid[0], firstMarg / firstCDF[-1])
	
	# sample from other dimensions
	for n in range(0, numbSampl):
		# piecew. poly. interpol in first dimension "Eq.~\ref{eq:LinPol}~\cite{dolgov2020approximation}"
		CurrApprCore = LinInterPol(SQTT[0], univarGrid[0], sampls[0][n])
		for d in range(1, dim):
			# calculate marginal PDF conditioned on previous samples
			# rank_min, gridSize, rank_pls = B[d].shape
			# MargDep = np.zeros((B[d].shape))
			# for r in range(0, rank_min):
			# # condition on previous samples
			#     MargDep[r, :, :] = CurrApprCore[0, r] * B[d][r, :, :]
			MargDep = np.einsum('lj,jbi->jbi', CurrApprCore, B[d])
				
			# calculate current 'conditional marginal' PDF as in "Eq.~\ref{eq:CurrMarg}~\cite[Eq.~31]{cui2022deep}"
			currMarg = gamError * lamX[d] + np.sum(np.sum(np.copy(MargDep), axis=0)** 2,
				axis=1) 
			 
			# normalise current PDF and compute CDF as in "Eq.~\ref{eq:CurrCDF} \cite[Eq.~17]{cui2022deep}"
			currCDF = scy.integrate.cumulative_trapezoid(currMarg, univarGrid[d], initial=0) 
			
			# draw sample as 'inverse transform'
			sampls[d][n] = np.interp(seeds[d][n], currCDF/currCDF[-1], univarGrid[d])
			probVal[d][n] = np.interp(sampls[d][n], univarGrid[d], currMarg/currCDF[-1])
			# piecew. poly. interpol., "Eq.~\ref{eq:LinPol}~\cite{dolgov2020approximation}", cond. on sampl. for next PDF
			CurrApprCore = np.copy(CurrApprCore) @ LinInterPol(SQTT[d], univarGrid[d], sampls[d][n]) 
	
		# approximated unnormalised function value may be useful for error analysis
		Approx[n] = gamError + CurrApprCore ** 2
	
	return sampls, probVal, Approx
\end{lstlisting}

\begin{lstlisting}[language=Python,	 numbers=none, columns=flexible, caption = {Python code to correct samples via the MH correction scheme.}, escapeinside={"}{"}]
def MHcorrection(samples, approx, trueLogFunc, probVals):
	accCount = 0
	rejCount = 0
	dim, numbSampl = samples.shape
	accProb = np.zeros(numbSampl - 1)
	correctSampl = np.zeros((dim, numbSampl))
	
	chainFuncTrueVal = np.zeros(numbSampl)
	chainFuncApproxVal = np.zeros(numbSampl)
	proposalFuncTrueVal = np.zeros(numbSampl)
	
	# initialise
	correctSampl[:, 0] = np.copy(samples[:, 0])
	
	currLogTarget = trueLogFunc(samples[:, 0])
	currLogTTDensity = np.sum(np.log(probVals[:, 0]))
	currTTApprox = np.copy(approx[0])
	
	chainFuncApproxVal[0] = currTTApprox
	chainFuncTrueVal[0] = np.exp(currLogTarget)
	proposalFuncTrueVal[0] = np.exp(currLogTarget)
	
	for i in range(1, numbSampl):
		propLogTarget = trueLogFunc(samples[:, i])
		propLogTTDensity = np.sum(np.log(probVals[:, i]))
		
		proposalFuncTrueVal[i] = np.exp(propLogTarget)
		
		MHRatioTT = currLogTTDensity - propLogTTDensity
		# alternatively
		# MHRatioTT = np.log(approx[i-1]) - np.log(approx[i])
		# MHRatioFunc = trueLogFunc(samples[:, i]) - trueLogFunc(correctSampl[:, i - 1])
		MHRatioFunc = propLogTarget - currLogTarget
		MHRatio = min(np.exp(MHRatioTT + MHRatioFunc), 1)
		accProb[i - 1] = MHRatio
		u = np.random.uniform()
		if u <= MHRatio:
			# accept
			accCount += 1
			correctSampl[:, i] = np.copy(samples[:, i])
			
			currLogTarget = propLogTarget
			currLogTTDensity = propLogTTDensity
			currTTApprox = np.copy(approx[i])
		else:
			# reject
			rejCount += 1
			correctSampl[:, i] = np.copy(correctSampl[:, i - 1])
		
		chainFuncApproxVal[i] = currTTApprox
		chainFuncTrueVal[i] = np.exp(currLogTarget)
	
	print(f'MH correction with {accCount} accepted and {rejCount} rejected proposal samples')
	
	return correctSampl, chainFuncApproxVal, chainFuncTrueVal, proposalFuncTrueVal, accProb

\end{lstlisting}

\begin{lstlisting}[language=Python,	 numbers=none, columns=flexible, caption = {Python code to calculate marginal PDFs as in Sec.~\ref{subsec:TTMarg}.}, escapeinside={"}{"}]
def getMargPDF(TTCore, univarGrid, absError):
	''' calculate marginal PDFs as in "Sec.~\ref{subsec:TTMarg} "using the coefficients from the forward marginalisation (see "Prop.~\ref{prob:ForMarg}") and backward marginalisation (see "Prop.~\ref{prob:backMarg}") '''

	dim = len(univarGrid)
	n =  len(univarGrid[0])
	D = [None] * dim
	margPDF = np.zeros((dim, n))
	
	# Lebesgue measure for quadrature "Eq.~\ref{eq:lebesgueWeight}"
	WholeLebLam = np.zeros(dim)
	for k in range(0, dim):
		WholeLebLam[k] = (univarGrid[k][-1] - univarGrid[k][0])
	
	lamX = np.ones(dim)
	for k in range(1, dim):
		lamX[k - 1] = np.prod(WholeLebLam[k:])
		
	# error as in "Eq.~\ref{eq:gamErr} \cite[Eq.~35]{cui2022deep}"
	gamError = absError / np.prod(WholeLebLam)
	
	# backward marginalisation as in "Prop.~\ref{prob:backMarg}"
	B = MargBack(TTCore, univarGrid)
	
	# forward marginalisation as in "Prop.~\ref{prob:ForMarg}"
	Rpre, Bpre = MargForw(TTCore, univarGrid)
	
	
	for k in range(0, dim):
		if k == 0:
			D[0] = B[0][0]
			# first marginal PDF "Eq.~\ref{eq:firstMarg} \cite[Eq.~30]{cui2022deep}"
			margPDF[0] = (gamError * np.prod(WholeLebLam[1:]) + np.sum(D[0] ** 2, 1))
		
		elif k == dim-1:
			# last marginal PDF "Eq.~\ref{eq:lastMarg}"
			margPDF[k] = gamError * np.prod(WholeLebLam[:k]) 
			"\phantom{margPDF[k] = gamError  }"  + np.sum(Bpre[k][:, :, 0] ** 2, 	0) * WholeLebLam[k]
		
		else:
			# other dimensions
			# r_kmin1, n, r_k = np.shape(B[k])
			# D[k] = np.zeros((r_kmin1, n, r_k))
			# for l_k in range(0, r_k):
			#     for l_kmin1 in range(0, r_kmin1):
			#         D[k][l_kmin1,:,l_k] = Rpre[k-1][l_kmin1,:] @ B[k][:, :, l_k]
			D[k] = np.einsum('lj,jbi->lbi', Rpre[k - 1], B[k])
			# marginal PDF as in "Eq.~\ref{eq:MargTT}"
			margPDF[k] = gamError * np.prod(WholeLebLam[k + 1:]) * np.prod(WholeLebLam[:k]) 
				"\phantom{margPDF[k] = gamError  }" + np.sum(np.sum(D[k] ** 2,0),1)  * WholeLebLam[k]
		
		# normalised marginal PDF
		margPDF[k] = margPDF[k] / np.sum(margPDF[k])
		
	return margPDF

\begin{lstlisting}[language=Python,	 numbers=none, columns=flexible, caption = {Python code for linear interpolation as in Eq.~\ref{eq:LinPol}.}, escapeinside={"}{"}]
def LinInterPol(TTCore,univarGrid, sampl):
	''' piecewise polynomial interpolation as in Dolgov et al." \cite{dolgov2020approximation}" '''
	
	# find index on grid
	idx = (np.abs(sampl - univarGrid)).argmin()
	if idx != 0 and univarGrid[idx-1] <= sampl and univarGrid[idx] >= sampl:
		idx = np.copy(idx)-1
	
	deNomX = univarGrid[idx+1] - univarGrid[idx]
	delX1 = (univarGrid[idx+1] - sampl )/deNomX
	delX2 = (sampl - univarGrid[idx]) /deNomX
	
	# interpolation as in "Eq.~\ref{eq:LinPol}"
	LinInterPolTTCore = TTCore[:,idx,:] * delX1 + TTCore[:,idx+1,:] * delX2
	
	return LinInterPolTTCore
\end{lstlisting}

\section{Randomise then Optimise -- Additional Information}
\label{sec:RTO_additional}
\begin{mccorrection} 
Recall we want to draw a sample from the full conditional 
\begin{equation*}
	\adjustbox{max width=0.99\textwidth}{$
		\bm{x} | \bm{\theta}, \delta, \gamma , \bm{y}\sim \mathcal{N}\Big(\underbrace{ \bm{\mu} + \big(\bm{Q}_{\delta} + \bm{A}_{\bm{\theta}}^T \bm{\Sigma}_{\gamma}^{-1} \bm{A}_{\bm{\theta}} \big)^{-1} \bm{A}_{\bm{\theta}}^T \bm{\Sigma}_{\gamma}^{-1} (\bm{y} - \bm{A}_{\bm{\theta}}\bm{\mu})}_{\bm{\mu}_{\bm{x} | \bm{\theta} , \delta, \gamma,  \bm{y}}},\underbrace{ \big(\bm{Q}_{\delta} + \bm{A}_{\bm{\theta}}^T \bm{\Sigma}_{\gamma}^{-1} \bm{A}_{\bm{\theta}}\big)^{-1}}_{\bm{\Sigma}_{\bm{x} | \bm{\theta} , \delta, \gamma,  \bm{y}}} \Big)\,$.}
\end{equation*}
Through the RTO method we obtain one independent sample $\bm{x}^{(k)} \sim \pi(\bm{x}|   \bm{\theta},  \delta, \gamma,  \bm{y})$ of the full conditional by solving the following equation:
\begin{align*}
	\bm{x}^{(k)} = \arg \min_{\bm{x}} \lVert \hat{\bm{A}} \bm{x} - ( \hat{\bm{y}} + \bm{b} ) \rVert_{L^2}^2  \, , \text{ where}\\
	\hat{\bm{A}} \coloneqq 
	\begin{bmatrix}
		\bm{\Sigma}_{\gamma}^{-1/2} \bm{A}_{\bm{\theta}} \ \\
		\bm{Q}_{\delta}^{1/2}
	\end{bmatrix}, \quad 
	\hat{\bm{y}} \coloneqq 
	\begin{bmatrix}
		\bm{\Sigma}_{\gamma}^{-1/2} \bm{y} \\
		\bm{Q}_{\delta}^{1/2} \bm{\mu}
	\end{bmatrix} , 
\end{align*}
$\bm{Q}_{\delta}$ is the prior precision, $\bm{\mu}$ the prior mean, $\bm{\Sigma}_{\gamma}$ the noise covariance  and $\bm{b} \sim \mathcal{N}(0,\bm{I})$ (see also~\cite{bardsley2014randomize,BardsleyTC2019RTO}).

In the following, we show that this optimisation problem indeed generates a sample from the desired conditional posterior distribution. 
We begin by considering $\lVert \hat{\bm{A}} \bm{x} - ( \hat{\bm{y}} + \bm{b} ) \rVert_{L^2}^2$, which we rewrite as:
\begin{align*}
	&\lVert \hat{\bm{A}} \bm{x} - ( \hat{\bm{y}} + \bm{b} ) \rVert_{L^2}^2 = \big( \hat{\bm{A}} \bm{x} - ( \hat{\bm{y}} + \bm{b} ) \big)^T  \big(	 \hat{\bm{A}} \bm{x} - ( \hat{\bm{y}} + \bm{b} ) \big)  \\
	&= \begin{bmatrix}
		\bm{\Sigma}_{\gamma}^{-1/2} \bm{A}_{\bm{\theta}} \bm{x} - 	\bm{\Sigma}_{\gamma}^{-1/2} \bm{y} - \tilde{\bm{v}}_1  \\
		\bm{Q}_{\delta}^{1/2} \bm{x}-	\bm{Q}_{\delta}^{1/2} \bm{\mu}-\tilde{\bm{v}}_2
	\end{bmatrix}^T 	 \begin{bmatrix}
		\bm{\Sigma}_{\gamma}^{-1/2} \bm{A}_{\bm{\theta}} \bm{x} - 	\bm{\Sigma}_{\gamma}^{-1/2} \bm{y} -\tilde{\bm{v}}_1  \\
		\bm{Q}_{\delta}^{1/2} \bm{x}-	\bm{Q}_{\delta}^{1/2}  \bm{\mu} - \tilde{\bm{v}}_2
	\end{bmatrix}\\
	&= \begin{bmatrix}
		\bm{x}^T \bm{A}_{\bm{\theta}}^T (\bm{\Sigma}_{\gamma}^{-1/2})^T - 	 \bm{y}^T (\bm{\Sigma}_{\gamma}^{-1/2} )^T- \tilde{\bm{v}}_1^T &
		\bm{x}^T	(\bm{Q}_{\delta}^{1/2})^T -	\bm{\mu}^T (\bm{Q}_{\delta}^{1/2})^T -\tilde{\bm{v}}_2^T
	\end{bmatrix}	 \begin{bmatrix}
		\bm{\Sigma}_{\gamma}^{-1/2} \bm{A}_{\bm{\theta}} \bm{x} - 	\bm{\Sigma}_{\gamma}^{-1/2} \bm{y} -\tilde{\bm{v}}_1\\
		\bm{Q}_{\delta}^{1/2} \bm{x}-	\bm{Q}_{\delta}^{1/2} \bm{\mu}-\tilde{\bm{v}}_2
	\end{bmatrix}\\
	&\begin{aligned}
		= \big[ \bm{x}^T \bm{A}_{\bm{\theta}}^T \bm{\Sigma}^{-1}_{\gamma} \bm{A}_{\bm{\theta}} \bm{x}   - 	 \bm{y}^T \bm{\Sigma}^{-1}_{\gamma} \bm{A}_{\bm{\theta}} \bm{x}  -\tilde{\bm{v}}_1^T	\bm{\Sigma}_{\gamma}^{-1/2}  \bm{A}_{\bm{\theta}} \bm{x} 
		-\bm{x}^T \bm{A}_{\bm{\theta}}^T \bm{\Sigma}^{-1}_{\gamma}  \bm{y}  +	 \bm{y}^T \bm{\Sigma}^{-1}_{\gamma}  \bm{y} +\tilde{\bm{v}}_1^T	\bm{\Sigma}_{\gamma}^{-1/2}   \bm{y}\\
		-\bm{x}^T \bm{A}_{\bm{\theta}}^T (\bm{\Sigma}_{\gamma}^{-1/2})^T \tilde{\bm{v}}_1    + 	 \bm{y}^T (\bm{\Sigma}_{\gamma}^{-1/2})^T \tilde{\bm{v}}_1  +\tilde{\bm{v}}_1^T\tilde{\bm{v}}_1
		+	\bm{x}^T	\bm{Q}_{\delta} \bm{x} -	\bm{\mu}^T \bm{Q}_{\delta}\bm{x} -\tilde{\bm{v}}_2^T \bm{Q}_{\delta} ^{1/2} \bm{x}\\
		-	\bm{x}^T	\bm{Q}_{\delta}  \bm{\mu} +	\bm{\mu}^T \bm{Q}_{\delta} \bm{\mu} + \tilde{\bm{v}}_2^T \bm{Q}_{\delta} ^{1/2}\bm{\mu}
		-	\bm{x}^T	(\bm{Q}_{\delta} ^{1/2})^T \tilde{\bm{v}}_2 +	\bm{\mu}^T (\bm{Q}_{\delta}^{1/2})^T \tilde{\bm{v}}_2 +\tilde{\bm{v}}_2^T\tilde{\bm{v}}_2 \big] 
	\end{aligned}
\end{align*}
with $\tilde{\bm{v}}_1 \sim \mathcal{N}(\bm{0},  \bm{I}_m)$ and $\tilde{\bm{v}}_2 \sim \mathcal{N}(\bm{0}, \bm{I}_n )$, since $\bm{y}\in \mathbb{R}^m$ and $\bm{x}\in \mathbb{R}^n$.
We take the derivative with respect to $\bm{x}$ and set to zero to determine $\bm{x}^{(k)} = \arg \min_{\bm{x}} \lVert \hat{\bm{A}} \bm{x} - ( \hat{\bm{y}} + \bm{b} ) \rVert_{L^2}^2 $.
\begin{align*}
	\begin{aligned}
		\big[ 		2 \bm{A}_{\bm{\theta}}^T \bm{\Sigma}^{-1}_{\gamma} \bm{A}_{\bm{\theta}} \bm{x}^{(k)}    - 	 \bm{y}^T \bm{\Sigma}^{-1}_{\gamma} \bm{A}_{\bm{\theta}}   -\tilde{\bm{v}}_1^T	\bm{\Sigma}_{\gamma}^{-1/2}  \bm{A}_{\bm{\theta}} 
		- \bm{A}_{\bm{\theta}}^T \bm{\Sigma}^{-1}_{\gamma}  \bm{y}  -	 
		\bm{A}_{\bm{\theta}}^T (\bm{\Sigma}_{\gamma}^{-1/2})^T \tilde{\bm{v}}_1  \\
		+ 	2	\bm{Q}_{\delta} \bm{x}^{(k)}  -	\bm{\mu}^T \bm{Q}_{\delta} -\tilde{\bm{v}}_2^T\bm{Q}_{\delta} ^{1/2} -		\bm{Q}_{\delta}  \bm{\mu} 
		-	(\bm{Q}_{\delta} ^{1/2})^T \tilde{\bm{v}}_2\big] = 0
	\end{aligned} 
\end{align*}
Observe that e.g. $(\bm{\Sigma}^{-1})^T = \bm{\Sigma}^{-1}$ and $ \bm{y}^T \bm{\Sigma}^{-1}_{\gamma} \bm{A}_{\bm{\theta}} +  \bm{A}_{\bm{\theta}}^T \bm{\Sigma}^{-1}_{\gamma}  \bm{y}  =  2\bm{A}_{\bm{\theta}}^T \bm{\Sigma}^{-1}_{\gamma}  \bm{y}  $.
Then we can rewrite
\begin{align*}
	\Leftrightarrow&&	0=& 2 \bm{A}_{\bm{\theta}}^T \bm{\Sigma}^{-1}_{\gamma} \bm{A}_{\bm{\theta}} \bm{x}^{(k)} 
	- 2 \bm{A}_{\bm{\theta}}^T \bm{\Sigma}^{-1}_{\gamma}  \bm{y}  -	 
	2 \bm{A}_{\bm{\theta}}^T \bm{\Sigma}_{\gamma}^{-1/2} \tilde{\bm{v}}_1    
	+2	\bm{Q}_{\delta} \bm{x}^{(k)}  -		2 \bm{Q}_{\delta}  \bm{\mu} 
	-	2	\bm{Q}_{\delta} ^{1/2} \tilde{\bm{v}}_2  \\
	\Leftrightarrow&&	0 =& \bm{A}_{\bm{\theta}}^T \bm{\Sigma}^{-1}_{\gamma} \bm{A}_{\bm{\theta}} \bm{x}^{(k)}  
	-  \bm{A}_{\bm{\theta}}^T \bm{\Sigma}^{-1}_{\gamma}  \bm{y}  
	-	 
	\bm{A}_{\bm{\theta}}^T \bm{\Sigma}_{\gamma}^{-1/2} \tilde{\bm{v}}_1   
	+
	\bm{Q}_{\delta} \bm{x}^{(k)}  -		 \bm{Q}_{\delta}  \bm{\mu} 
	-	\bm{Q}_{\delta} ^{1/2} \tilde{\bm{v}}_2  \\
	\Leftrightarrow&&		\bm{A}_{\bm{\theta}}^T \bm{\Sigma}^{-1}_{\gamma} \bm{A}_{\bm{\theta}} \bm{x}^{(k)} 
	+ \bm{Q}_{\delta} \bm{x}^{(k)}
	=& 	\bm{Q}_{\delta} ^{1/2} \tilde{\bm{v}}_2  +
	\bm{A}_{\bm{\theta}}^T \bm{\Sigma}_{\gamma}^{-1/2} \tilde{\bm{v}}_2  +		 \bm{Q}_{\delta}  \bm{\mu} + \bm{A}_{\bm{\theta}}^T \bm{\Sigma}^{-1}_{\gamma}  \bm{y}     \\
	\Leftrightarrow&&  (\bm{A}_{\bm{\theta}}^T \bm{\Sigma}^{-1}_{\gamma} \bm{A}_{\bm{\theta}} 
	+ \bm{Q}_{\delta} )\bm{x}^{(k)} 
	=& 	 	 \bm{Q}_{\delta}  \bm{\mu} + \bm{A}_{\bm{\theta}}^T \bm{\Sigma}^{-1}_{\gamma}  \bm{y}  +\underbrace{\bm{Q}_{\delta} ^{1/2} \tilde{\bm{v}}_2}_{\sim \mathcal{N}(0,\bm{Q}_{\delta})}  +
	\underbrace{ \bm{A}_{\bm{\theta}}^T \bm{\Sigma}_{\gamma}^{-1/2} \tilde{\bm{v}}_1 }_{\sim \mathcal{N}(0,\bm{A}_{\bm{\theta}}^T \bm{\Sigma}_{\gamma}^{-1}\bm{A}_{\bm{\theta}})}   \\
	\Leftrightarrow&&	(\bm{A}_{\bm{\theta}}^T \bm{\Sigma}^{-1}_{\gamma} \bm{A}_{\bm{\theta}} 
	+ \bm{Q}_{\delta} )\bm{x}^{(k)}
	=& 	 	 \bm{Q}_{\delta}  \bm{\mu} + \bm{A}_{\bm{\theta}}^T \bm{\Sigma}^{-1}_{\gamma}  \bm{y}  +\bm{w}  \, , \text{with } \bm{w} \sim \mathcal{N}(0,\bm{A}_{\bm{\theta}}^T \bm{\Sigma}^{-1}_{\gamma} \bm{A}_{\bm{\theta}} 
	+ \bm{Q}_{\delta}) \, .
\end{align*}
Furthermore,
\begin{align*}
	\bm{x}^{(k)}
	= 	 	(\bm{A}_{\bm{\theta}}^T \bm{\Sigma}^{-1}_{\gamma} \bm{A}_{\bm{\theta}} 
	+ \bm{Q}_{\delta} )^{-1}(\bm{Q}_{\delta}  \bm{\mu} + \bm{A}_{\bm{\theta}}^T \bm{\Sigma}^{-1}_{\gamma}  \bm{y})  + \tilde{\bm{w} } \, , \text{with } \tilde{\bm{w} } \sim \mathcal{N}(0,(\bm{A}_{\bm{\theta}}^T \bm{\Sigma}^{-1}_{\gamma} \bm{A}_{\bm{\theta}} 
	+ \bm{Q}_{\delta} )^{-1}) \, .
\end{align*}
Finally we show that
\begin{align*}
	\bm{x}^{(k)} 	&= 	 	(\bm{A}_{\bm{\theta}}^T \bm{\Sigma}^{-1}_{\gamma} \bm{A}_{\bm{\theta}} 
	+ \bm{Q}_{\delta} )^{-1}(\bm{Q}_{\delta}  \bm{\mu}+ \bm{A}_{\bm{\theta}}^T \bm{\Sigma}_{\gamma}^{-1}  \bm{A}_{\bm{\theta}} \bm{\mu}- \bm{A}_{\bm{\theta}}^T \bm{\Sigma}_{\gamma}^{-1}  \bm{A}_{\bm{\theta}} \bm{\mu} + \bm{A}_{\bm{\theta}}^T \bm{\Sigma}^{-1}_{\gamma}  \bm{y})  + \tilde{\bm{w} }\\
